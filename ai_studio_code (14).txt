-- [[ FILTRO DE EQUIPO Y DAÑO VÁLIDO ]]
local function isTeammate(player)
    if not player or player == LocalPlayer then return true end
    if player == protectedPlayer then return true end

    -- Comprobación de equipo (Team / TeamColor)
    if not player.Neutral and not LocalPlayer.Neutral then
        if player.Team and LocalPlayer.Team and player.Team == LocalPlayer.Team then
            return true
        end
        if player.TeamColor and LocalPlayer.TeamColor and player.TeamColor == LocalPlayer.TeamColor then
            return true
        end
    end
    return false
end

local function canDamage(player, char)
    if not player or not char then return false end
    if isTeammate(player) then return false end

    -- Ignorar si tiene escudo activo (ForceField)
    if char:FindFirstChildWhichIsA("ForceField") then
        return false
    end

    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then
        return false
    end

    return true
end

local function applySpeedState(enable)
    speedEnabled = enable
    if speedConn then speedConn:Disconnect() speedConn = nil end

    if speedEnabled then
        BtnToggleSpeed.Text = "SPEED: [ON]"
        BtnToggleSpeed.BackgroundColor3 = Color3.fromRGB(255, 200, 0)

        speedConn = RunService.Heartbeat:Connect(function()
            if not speedEnabled then return end
            local char = LocalPlayer.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if hum and hum.WalkSpeed ~= targetSpeed then
                hum.WalkSpeed = targetSpeed
            end
        end)
    else
        BtnToggleSpeed.Text = "SPEED: [OFF]"
        BtnToggleSpeed.BackgroundColor3 = Color3.fromRGB(35, 36, 42)

        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then hum.WalkSpeed = 16 end
    end
end

BtnToggleSpeed.MouseButton1Click:Connect(function()
    applySpeedState(not speedEnabled)
end)

local function getWeapon()
    local char = LocalPlayer.Character
    local bp = LocalPlayer:FindFirstChild("Backpack")
    if not char then return nil end

    local names = {"Knife", "Sword", "LinkedSword", "ClassicSword"}
    for _, n in ipairs(names) do
        local tool = char:FindFirstChild(n) or (bp and bp:FindFirstChild(n))
        if tool then return tool end
    end

    for _, v in ipairs(char:GetChildren()) do if v:IsA("Tool") then return v end end
    if bp then for _, v in ipairs(bp:GetChildren()) do if v:IsA("Tool") then return v end end end
    return nil
end

local function executeInstaKill(targetChar)
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 or not targetChar then return end

    local weapon = getWeapon()
    if not weapon then return end

    if weapon.Parent == LocalPlayer:FindFirstChild("Backpack") then
        hum:EquipTool(weapon)
        task.wait(0.01)
    end

    if weapon:FindFirstChild("Stab") then
        weapon.Stab:FireServer("Slash")
    end

    weapon:Activate()

    local handle = weapon:FindFirstChild("Handle") or weapon:FindFirstChildWhichIsA("BasePart")
    if handle and typeof(firetouchinterest) == "function" then
        local tHRP = targetChar:FindFirstChild("HumanoidRootPart") or targetChar:FindFirstChild("Torso") or targetChar:FindFirstChild("UpperTorso")
        if tHRP then
            pcall(function()
                firetouchinterest(tHRP, handle, 0)
                firetouchinterest(tHRP, handle, 1)
            end)
        end
    end
end

local function stopLock()
    isLockedFront = false
    BtnKillAction.Text = "KILL PLAYER"
    BtnKillAction.BackgroundColor3 = Color3.fromRGB(210, 30, 50)

    if lockConn then lockConn:Disconnect() lockConn = nil end
    if noclipConn then noclipConn:Disconnect() noclipConn = nil end

    for target, _ in pairs(selectedKillPlayers) do
        if target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
            target.Character.HumanoidRootPart.AssemblyLinearVelocity = Vector3.zero
            target.Character.HumanoidRootPart.AssemblyAngularVelocity = Vector3.zero
        end
    end
end

local function startLockAndKill()
    isLockedFront = true
    BtnKillAction.Text = "PURGANDO..."
    BtnKillAction.BackgroundColor3 = Color3.fromRGB(0, 140, 255)

    if noclipConn then noclipConn:Disconnect() end
    noclipConn = RunService.Stepped:Connect(function()
        if not isLockedFront then return end
        for target, isSel in pairs(selectedKillPlayers) do
            if isSel and target.Character then
                for _, part in ipairs(target.Character:GetDescendants()) do
                    if part:IsA("BasePart") then part.CanCollide = false end
                end
            end
        end
    end)

    if lockConn then lockConn:Disconnect() end
    lockConn = RunService.Heartbeat:Connect(function()
        local char = LocalPlayer.Character
        local myHRP = char and char:FindFirstChild("HumanoidRootPart")
        if not myHRP then return end

        local count = 0
        for target, isSel in pairs(selectedKillPlayers) do
            if isSel and target ~= LocalPlayer and target.Character and canDamage(target, target.Character) then
                local tChar = target.Character
                local tHRP = tChar:FindFirstChild("HumanoidRootPart")
                local tHum = tChar:FindFirstChildOfClass("Humanoid")

                if tHRP and tHum and tHum.Health > 0 then
                    count = count + 1
                    local offset = (count - 1) * 1.2
                    tHRP.CFrame = myHRP.CFrame * CFrame.new(offset, 0, -2.5)
                    tHRP.AssemblyLinearVelocity = Vector3.zero
                    tHRP.AssemblyAngularVelocity = Vector3.zero
                    executeInstaKill(tChar)
                end
            end
        end
    end)
end

BtnKillAction.MouseButton1Click:Connect(startLockAndKill)
BtnReleaseAction.MouseButton1Click:Connect(stopLock)

-- Bucle: Mi Círculo (Ignora equipo y jugadores inmunes)
local selfDebounce = false
local function runSelfGuardScan()
    if selfGuardLoop then task.cancel(selfGuardLoop) selfGuardLoop = nil end
    selfGuardLoop = task.spawn(function()
        while selfGuardEnabled do
            task.wait(0.03)
            local myChar = LocalPlayer.Character
            local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")

            if myHRP and not selfDebounce then
                for _, player in ipairs(Players:GetPlayers()) do
                    if player.Character and canDamage(player, player.Character) then
                        local enemyChar = player.Character
                        local enemyHRP = enemyChar:FindFirstChild("HumanoidRootPart")
                        local enemyHum = enemyChar:FindFirstChildOfClass("Humanoid")

                        if enemyHRP and enemyHum and enemyHum.Health > 0 then
                            local dist = (myHRP.Position - enemyHRP.Position).Magnitude
                            if dist <= selfRadius then
                                selfDebounce = true

                                for _, p in ipairs(enemyChar:GetDescendants()) do
                                    if p:IsA("BasePart") then p.CanCollide = false end
                                end

                                enemyHRP.CFrame = myHRP.CFrame * CFrame.new(0, 0, -2.5)
                                enemyHRP.AssemblyLinearVelocity = Vector3.zero

                                executeInstaKill(enemyChar)

-- Bucle: Proximity Target (Ignora equipo del protegido y jugadores inmunes)
local targetDebounce = false
local function runTargetGuardScan()
    if playerGuardLoop then task.cancel(playerGuardLoop) playerGuardLoop = nil end
    playerGuardLoop = task.spawn(function()
        while playerGuardEnabled do
            task.wait(0.04)
            if protectedPlayer and protectedPlayer.Character and LocalPlayer.Character then
                local protHRP = protectedPlayer.Character:FindFirstChild("HumanoidRootPart")
                local myHRP = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")

                if protHRP and myHRP and not targetDebounce then
                    for _, player in ipairs(Players:GetPlayers()) do
                        if player.Character and canDamage(player, player.Character) then
                            local enemyChar = player.Character
                            local enemyHRP = enemyChar:FindFirstChild("HumanoidRootPart")
                            local enemyHum = enemyChar:FindFirstChildOfClass("Humanoid")

                            if enemyHRP and enemyHum and enemyHum.Health > 0 then
                                local dist = (protHRP.Position - enemyHRP.Position).Magnitude
                                if dist <= playerRadius then
                                    targetDebounce = true

                                    for _, p in ipairs(enemyChar:GetDescendants()) do
                                        if p:IsA("BasePart") then p.CanCollide = false end
                                    end

                                    enemyHRP.CFrame = myHRP.CFrame * CFrame.new(0, 0, -2.5)
                                    enemyHRP.AssemblyLinearVelocity = Vector3.zero

                                    executeInstaKill(enemyChar)
                                    task.wait(0.08)
                                    targetDebounce = false
                                    break
                                end
                            end
                        end
                    end
                end
            end
        end
    end)
end

local function stopSpectate()
    isSpectating = false
    if spectateConn then spectateConn:Disconnect() spectateConn = nil end

    local myHum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
    if myHum then
        Camera.CameraSubject = myHum
    end
    BtnSpectate.Text = "SPECTATE"
    BtnSpectate.BackgroundColor3 = Color3.fromRGB(130, 40, 210)
end

local function startSpectate()
    if not spectateTarget or not spectateTarget.Character then
        BtnSpectate.Text = "¡ELIGE JUGADOR!"
        task.delay(1, function() BtnSpectate.Text = "SPECTATE" end)
        return
    end

    local tHum = spectateTarget.Character:FindFirstChildOfClass("Humanoid")
    if not tHum or tHum.Health <= 0 then
        stopSpectate()
        return
    end

    isSpectating = true
    Camera.CameraSubject = tHum
    BtnSpectate.Text = "VIENDO: " .. string.sub(spectateTarget.DisplayName, 1, 8)
    BtnSpectate.BackgroundColor3 = Color3.fromRGB(0, 190, 100)

    if spectateConn then spectateConn:Disconnect() end
    spectateConn = RunService.RenderStepped:Connect(function()
        if not isSpectating then
            if spectateConn then spectateConn:Disconnect() end
            return
        end

        if not spectateTarget or not spectateTarget.Parent or not spectateTarget.Character then
            stopSpectate()
            return
        end

        local hum = spectateTarget.Character:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then
            stopSpectate()
            return
        end

        if Camera.CameraSubject ~= hum then
            Camera.CameraSubject = hum
        end
    end)
end

BtnSpectate.MouseButton1Click:Connect(startSpectate)
BtnUnspectate.MouseButton1Click:Connect(stopSpectate)

local kBtns = {}
local pBtns = {}
local sBtns = {}

local function updateAllLists()
    for _, b in pairs(kBtns) do b:Destroy() end
    for _, b in pairs(pBtns) do b:Destroy() end
    for _, b in pairs(sBtns) do b:Destroy() end
    table.clear(kBtns)
    table.clear(pBtns)
    table.clear(sBtns)

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            -- Botón Kill
            local kb = Instance.new("TextButton")
            kb.Size = UDim2.new(1, 0, 0, 22)
            kb.Text = " " .. player.DisplayName
            kb.Font = Enum.Font.Gotham
            kb.TextSize = 9
            kb.TextXAlignment = Enum.TextXAlignment.Left
            kb.BorderSizePixel = 0
            kb.AutoButtonColor = false
            kb.Parent = ScrollKill
            Instance.new("UICorner", kb).CornerRadius = UDim.new(0, 4)

            if selectedKillPlayers[player] then
                kb.BackgroundColor3 = Color3.fromRGB(0, 140, 255)
                kb.TextColor3 = Color3.fromRGB(255, 255, 255)
            else
                kb.BackgroundColor3 = Color3.fromRGB(20, 21, 26)
                kb.TextColor3 = Color3.fromRGB(180, 180, 190)
            end

            kb.MouseButton1Click:Connect(function()
                if selectedKillPlayers[player] then
                    selectedKillPlayers[player] = nil
                    kb.BackgroundColor3 = Color3.fromRGB(20, 21, 26)
                    kb.TextColor3 = Color3.fromRGB(180, 180, 190)
                else
                    selectedKillPlayers[player] = true
                    kb.BackgroundColor3 = Color3.fromRGB(0, 140, 255)
                    kb.TextColor3 = Color3.fromRGB(255, 255, 255)
                end
            end)
            kBtns[player] = kb

            -- Botón Proximity
            local pb = Instance.new("TextButton")
            pb.Size = UDim2.new(1, 0, 0, 22)
            pb.Text = " " .. player.DisplayName
            pb.Font = Enum.Font.Gotham
            pb.TextSize = 9
            pb.TextXAlignment = Enum.TextXAlignment.Left
            pb.BorderSizePixel = 0
            pb.AutoButtonColor = false
            pb.Parent = ScrollProx
            Instance.new("UICorner", pb).CornerRadius = UDim.new(0, 4)

            if protectedPlayer == player then
                pb.BackgroundColor3 = Color3.fromRGB(0, 210, 255)
                pb.TextColor3 = Color3.fromRGB(255, 255, 255)
            else
                pb.BackgroundColor3 = Color3.fromRGB(20, 21, 26)
                pb.TextColor3 = Color3.fromRGB(180, 180, 190)
            end

            pb.MouseButton1Click:Connect(function()
                if protectedPlayer == player then
                    protectedPlayer = nil
                else
                    protectedPlayer = player
                end
                updateAllLists()
            end)
            pBtns[player] = pb

            -- Botón Spectate
            local sb = Instance.new("TextButton")
            sb.Size = UDim2.new(1, 0, 0, 22)
            sb.Text = " " .. player.DisplayName
            sb.Font = Enum.Font.Gotham
            sb.TextSize = 9
            sb.TextXAlignment = Enum.TextXAlignment.Left
            sb.BorderSizePixel = 0
            sb.AutoButtonColor = false
            sb.Parent = ScrollSpec
            Instance.new("UICorner", sb).CornerRadius = UDim.new(0, 4)

            if spectateTarget == player then
                sb.BackgroundColor3 = Color3.fromRGB(160, 60, 255)
                sb.TextColor3 = Color3.fromRGB(255, 255, 255)
            else
                sb.BackgroundColor3 = Color3.fromRGB(20, 21, 26)
                sb.TextColor3 = Color3.fromRGB(180, 180, 190)
            end

            sb.MouseButton1Click:Connect(function()
                if spectateTarget == player then
                    spectateTarget = nil
                    if isSpectating then stopSpectate() end
                else
                    spectateTarget = player
                    if isSpectating then startSpectate() end
                end
                updateAllLists()
            end)
            sBtns[player] = sb
        end
    end
end
