local function getWeapon()
    local char = LocalPlayer.Character
    local bp = LocalPlayer:FindFirstChild("Backpack")
    if not char then return nil end

    local names = {"Knife", "Sword", "LinkedSword", "ClassicSword"}
    for _, n in ipairs(names) do
        local tool = char:FindFirstChild(n) or (bp and bp:FindFirstChild(n))
        if tool then return tool end
    end

    for _, v in ipairs(char:GetChildren()) do if v:IsA("Tool") then return v end end
    if bp then for _, v in ipairs(bp:GetChildren()) do if v:IsA("Tool") then return v end end end
    return nil
end

local function executeInstaKill(targetChar)
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 or not targetChar then return end

    local weapon = getWeapon()
    if not weapon then return end

    if weapon.Parent == LocalPlayer:FindFirstChild("Backpack") then
        hum:EquipTool(weapon)
        task.wait(0.01)
    end

    -- 1. MM2 Remote
    if weapon:FindFirstChild("Stab") then
        weapon.Stab:FireServer("Slash")
    end

    -- 2. Espadazo nativo
    weapon:Activate()

    -- 3. Toque de daño forzado por hitbox directo
    local handle = weapon:FindFirstChild("Handle") or weapon:FindFirstChildWhichIsA("BasePart")
    if handle and typeof(firetouchinterest) == "function" then
        local tHRP = targetChar:FindFirstChild("HumanoidRootPart") or targetChar:FindFirstChild("Torso") or targetChar:FindFirstChild("UpperTorso")
        if tHRP then
            pcall(function()
                firetouchinterest(tHRP, handle, 0)
                firetouchinterest(tHRP, handle, 1)
            end)
        end
    end
end

local function stopLock()
    isLockedFront = false
    BtnKillAction.Text = "KILL PLAYER"
    BtnKillAction.BackgroundColor3 = Color3.fromRGB(210, 30, 50)

    if lockConn then lockConn:Disconnect() lockConn = nil end
    if noclipConn then noclipConn:Disconnect() noclipConn = nil end

    for target, _ in pairs(selectedKillPlayers) do
        if target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
            target.Character.HumanoidRootPart.AssemblyLinearVelocity = Vector3.zero
            target.Character.HumanoidRootPart.AssemblyAngularVelocity = Vector3.zero
        end
    end
end

local function startLockAndKill()
    isLockedFront = true
    BtnKillAction.Text = "PURGANDO..."
    BtnKillAction.BackgroundColor3 = Color3.fromRGB(0, 140, 255)

    -- Quitar colisiones en Stepped para caminar sin estorbos
    if noclipConn then noclipConn:Disconnect() end
    noclipConn = RunService.Stepped:Connect(function()
        if not isLockedFront then return end
        for target, isSel in pairs(selectedKillPlayers) do
            if isSel and target.Character then
                for _, part in ipairs(target.Character:GetDescendants()) do
                    if part:IsA("BasePart") then part.CanCollide = false end
                end
            end
        end
    end)

    -- Pegar al frente y matar
    if lockConn then lockConn:Disconnect() end
    lockConn = RunService.Heartbeat:Connect(function()
        local char = LocalPlayer.Character
        local myHRP = char and char:FindFirstChild("HumanoidRootPart")
        if not myHRP then return end

        local count = 0
        for target, isSel in pairs(selectedKillPlayers) do
            if isSel and target ~= LocalPlayer and target.Character then
                local tChar = target.Character
                local tHRP = tChar:FindFirstChild("HumanoidRootPart")
                local tHum = tChar:FindFirstChildOfClass("Humanoid")

                if tHRP and tHum and tHum.Health > 0 then
                    count = count + 1
                    local offset = (count - 1) * 1.2
                    tHRP.CFrame = myHRP.CFrame * CFrame.new(offset, 0, -2.5)
                    tHRP.AssemblyLinearVelocity = Vector3.zero
                    tHRP.AssemblyAngularVelocity = Vector3.zero
                    executeInstaKill(tChar)
                end
            end
        end
    end)
end

BtnKillAction.MouseButton1Click:Connect(startLockAndKill)
BtnReleaseAction.MouseButton1Click:Connect(stopLock)

-- Actualización de la posición del círculo en el suelo
RunService.RenderStepped:Connect(function()
    if guardEnabled and protectedPlayer and protectedPlayer.Character then
        local targetHRP = protectedPlayer.Character:FindFirstChild("HumanoidRootPart")
        if targetHRP then
            circleAdornment.Visible = true
            circleAdornment.Adornee = targetHRP
            circleAdornment.Radius = guardRadius
            return
        end
    end
    circleAdornment.Visible = false
    circleAdornment.Adornee = nil
end)

local guardDebounce = false

local function runProximityScan()
    if guardLoop then task.cancel(guardLoop) guardLoop = nil end

    guardLoop = task.spawn(function()
        while guardEnabled do
            task.wait(0.04)
            if protectedPlayer and protectedPlayer.Character and LocalPlayer.Character then
                local protHRP = protectedPlayer.Character:FindFirstChild("HumanoidRootPart")
                local myHRP = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")

                if protHRP and myHRP and not guardDebounce then
                    for _, player in ipairs(Players:GetPlayers()) do
                        -- Si es otro jugador (enemigo que NO eres tú ni el protegido)
                        if player ~= LocalPlayer and player ~= protectedPlayer and player.Character then
                            local enemyChar = player.Character
                            local enemyHRP = enemyChar:FindFirstChild("HumanoidRootPart")
                            local enemyHum = enemyChar:FindFirstChildOfClass("Humanoid")

                            if enemyHRP and enemyHum and enemyHum.Health > 0 then
                                local dist = (protHRP.Position - enemyHRP.Position).Magnitude
                                if dist <= guardRadius then
                                    guardDebounce = true

                                    -- Apagar colisiones del intruso
                                    for _, p in ipairs(enemyChar:GetDescendants()) do
                                        if p:IsA("BasePart") then p.CanCollide = false end
                                    end

                                    -- Ponerlo al frente de nosotros y reventarlo al instante
                                    enemyHRP.CFrame = myHRP.CFrame * CFrame.new(0, 0, -2.5)
                                    enemyHRP.AssemblyLinearVelocity = Vector3.zero

                                    executeInstaKill(enemyChar)
                                    task.wait(0.1)
                                    guardDebounce = false
                                    break
                                end
                            end
                        end
                    end
                end
            end
        end
    end)
end

BtnToggleGuard.MouseButton1Click:Connect(function()
    guardEnabled = not guardEnabled
    if guardEnabled then
        BtnToggleGuard.Text = "ACTIVAR GUARDIA: [ON]"
        BtnToggleGuard.BackgroundColor3 = Color3.fromRGB(0, 180, 255)

    else
        BtnToggleGuard.Text = "ACTIVAR GUARDIA: [OFF]"
        BtnToggleGuard.BackgroundColor3 = Color3.fromRGB(35, 36, 42)
        if guardLoop then task.cancel(guardLoop) guardLoop = nil end
        circleAdornment.Visible = false
    end
end)

local killButtons = {}
local proxButtons = {}

local function updateAllLists()
    for _, b in pairs(killButtons) do b:Destroy() end
    for _, b in pairs(proxButtons) do b:Destroy() end
    table.clear(killButtons)
    table.clear(proxButtons)

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            -- 1. Botón para Lista de Purga
            local kBtn = Instance.new("TextButton")
            kBtn.Size = UDim2.new(1, 0, 0, 22)
            kBtn.Text = " " .. player.DisplayName
            kBtn.Font = Enum.Font.Gotham
            kBtn.TextSize = 10
            kBtn.TextXAlignment = Enum.TextXAlignment.Left
            kBtn.BorderSizePixel = 0
            kBtn.AutoButtonColor = false
            kBtn.Parent = ScrollKill
            Instance.new("UICorner", kBtn).CornerRadius = UDim.new(0, 4)

            if selectedKillPlayers[player] then
                kBtn.BackgroundColor3 = Color3.fromRGB(0, 140, 255)
                kBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
            else
                kBtn.BackgroundColor3 = Color3.fromRGB(20, 21, 26)
                kBtn.TextColor3 = Color3.fromRGB(180, 180, 190)
            end

            kBtn.MouseButton1Click:Connect(function()
                if selectedKillPlayers[player] then
                    selectedKillPlayers[player] = nil
                    kBtn.BackgroundColor3 = Color3.fromRGB(20, 21, 26)
                    kBtn.TextColor3 = Color3.fromRGB(180, 180, 190)
                else
                    selectedKillPlayers[player] = true
                    kBtn.BackgroundColor3 = Color3.fromRGB(0, 140, 255)
                    kBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
                end
            end)
            killButtons[player] = kBtn

            -- 2. Botón para Proximity Player (Protegido)
            local pBtn = Instance.new("TextButton")
            pBtn.Size = UDim2.new(1, 0, 0, 22)
            pBtn.Text = " " .. player.DisplayName
            pBtn.Font = Enum.Font.Gotham
            pBtn.TextSize = 10
            pBtn.TextXAlignment = Enum.TextXAlignment.Left
            pBtn.BorderSizePixel = 0
            pBtn.AutoButtonColor = false
            pBtn.Parent = ScrollProx
            Instance.new("UICorner", pBtn).CornerRadius = UDim.new(0, 4)

            if protectedPlayer == player then
                pBtn.BackgroundColor3 = Color3.fromRGB(0, 220, 120)
                pBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
            else
                pBtn.BackgroundColor3 = Color3.fromRGB(20, 21, 26)
                pBtn.TextColor3 = Color3.fromRGB(180, 180, 190)
            end

            pBtn.MouseButton1Click:Connect(function()
                if protectedPlayer == player then
                    protectedPlayer = nil
                else
                    protectedPlayer = player
                end
                updateAllLists()
            end)
            proxButtons[player] = pBtn
        end
    end
end

Players.PlayerAdded:Connect(function() task.wait(0.2) updateAllLists() end)
Players.PlayerRemoving:Connect(function(player)
    selectedKillPlayers[player] = nil
    if protectedPlayer == player then protectedPlayer = nil end
    updateAllLists()
end)

switchTab(true)
updateAllLists()
